cmake_minimum_required(VERSION 3.20)
project(drivebrain_uart_bench LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(POLICY CMP0167)
    cmake_policy(SET CMP0167 NEW)
endif()
find_package(Boost REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(Protobuf CONFIG QUIET)
if(NOT Protobuf_FOUND)
    find_package(Protobuf REQUIRED)
endif()
find_package(spdlog REQUIRED)
find_package(Threads REQUIRED)
set(COMMS "${CMAKE_CURRENT_LIST_DIR}/../../../drivebrain_comms")
add_library(uart_driver "${COMMS}/src/UARTComms.cpp")
target_include_directories(uart_driver PUBLIC "${COMMS}/include")
target_link_libraries(uart_driver PUBLIC Boost::headers OpenSSL::Crypto protobuf::libprotobuf spdlog::spdlog Threads::Threads)

set(HT_PROTO_DIR "" CACHE PATH "Pinned HT_proto source directory")
if(NOT EXISTS "${HT_PROTO_DIR}/hytech_msgs.proto")
    message(FATAL_ERROR "Set -DHT_PROTO_DIR=/path/to/HT_proto/proto (use the same revision on Pi and Teensy)")
endif()
set(GENERATED "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(MAKE_DIRECTORY "${GENERATED}")
add_custom_command(
    OUTPUT "${GENERATED}/base_msgs.pb.cc" "${GENERATED}/base_msgs.pb.h"
           "${GENERATED}/hytech_msgs.pb.cc" "${GENERATED}/hytech_msgs.pb.h"
           "${GENERATED}/uart_bench.pb.cc" "${GENERATED}/uart_bench.pb.h"
    COMMAND protobuf::protoc "--cpp_out=${GENERATED}" "-I${HT_PROTO_DIR}" "-I${CMAKE_CURRENT_LIST_DIR}"
            "${HT_PROTO_DIR}/base_msgs.proto" "${HT_PROTO_DIR}/hytech_msgs.proto" "${CMAKE_CURRENT_LIST_DIR}/uart_bench.proto"
    DEPENDS "${HT_PROTO_DIR}/base_msgs.proto" "${HT_PROTO_DIR}/hytech_msgs.proto" uart_bench.proto
    VERBATIM)
add_library(uart_bench_messages "${GENERATED}/base_msgs.pb.cc" "${GENERATED}/hytech_msgs.pb.cc" "${GENERATED}/uart_bench.pb.cc")
target_include_directories(uart_bench_messages PUBLIC "${GENERATED}")
target_link_libraries(uart_bench_messages PUBLIC protobuf::libprotobuf)
add_executable(uart_hil_test uart_hil_test.cpp)
target_link_libraries(uart_hil_test PRIVATE uart_driver uart_bench_messages)
add_executable(uart_command_test uart_command_test.cpp)
target_link_libraries(uart_command_test PRIVATE uart_driver uart_bench_messages)
enable_testing()
add_test(NAME uart_command_validation COMMAND uart_command_test)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME uart_virtual_serial COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/test_virtual_serial.py" "$<TARGET_FILE:uart_hil_test>")
set_tests_properties(uart_virtual_serial PROPERTIES TIMEOUT 90)
